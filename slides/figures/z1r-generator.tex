% Edge-emitting Zero--One--Random generator with Bernoulli(p) random phase.
% Editable diagram; labels have the form emitted symbol : edge probability.
\begin{tikzpicture}
  \node[generator state] (zero) at (0,0) {$s_0$};
  \node[generator state] (one) at (4.8,0) {$s_1$};
  \node[generator state] (random) at (2.4,-1.55) {$s_R$};
  \draw[generator edge] (zero) -- node[emission label] {$0:1$} (one);
  \draw[generator edge] (one) -- node[emission label,sloped] {$1:1$} (random);
  \draw[generator edge] (random) to[bend left=27] node[emission label,sloped] {$0:1-p$} (zero);
  \draw[generator edge] (random) to[bend right=27] node[emission label,sloped] {$1:p$} (zero);
\end{tikzpicture}
